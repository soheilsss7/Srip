name: rasmio graph v3

# ربات همیشگی رسمیو — سه راه اجرا:
#   · schedule: هر روز خودکار (۰۱:۳۰ بامداد تهران) — بدون هیچ دخالتی
#   · repository_dispatch (rasmio-run): عامل Arena هر وقت خواست اجرا می‌کند
#   · workflow_dispatch: دکمهٔ دستی
# نشست ماندگار: کوکی‌ها با کلید Secret (RASMIO_SESSION_KEY) رمز می‌شوند و
# در out/session.enc ذخیره می‌شوند؛ اجراهای بعدی بدون پیامک.
on:
  repository_dispatch:
    types: [rasmio-run]
  schedule:
    - cron: '0 22 * * *'
  workflow_dispatch: {}

permissions:
  contents: write

jobs:
  capture:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    env:
      RASMIO_PHONE: ${{ secrets.RASMIO_PHONE }}
      RASMIO_SESSION_KEY: ${{ secrets.RASMIO_SESSION_KEY }}
      OTP_BRANCH: arena/01a08a50-srip
    steps:
      - uses: actions/checkout@v4
        with:
          ref: arena/01a08a50-srip
          fetch-depth: 10
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - name: install
        working-directory: scripts/rasmio
        run: npm install --no-audit --no-fund
      - name: run robot
        working-directory: scripts/rasmio
        run: |
          mkdir -p out shots
          node capture.mjs 2>&1 | tee out/run.log
      - name: commit output
        if: always()
        run: |
          git config user.name "rasmio-bot"
          git config user.email "actions@users.noreply.github.com"
          git add -f scripts/rasmio/out scripts/rasmio/shots scripts/rasmio/data scripts/rasmio/frontier.json || true
          git commit -m "rasmio: crawl $(date -u +%Y.%m.%d-%H%M)" || true
          for i in 1 2 3; do
            git fetch origin arena/01a08a50-srip || true
            git rebase origin/arena/01a08a50-srip || (git rebase --abort || true)
            if git push origin HEAD:arena/01a08a50-srip; then echo "[bot] push OK"; exit 0; fi
            sleep 5
          done
          echo "[bot] push failed after retries - artifact exists"
      - name: save output
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: rasmio-capture
          retention-days: 1
          path: |
            scripts/rasmio/out/
            scripts/rasmio/shots/
